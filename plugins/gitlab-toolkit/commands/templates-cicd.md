# GitLab CI/CD Templates

프로젝트 타입에 맞는 `.gitlab-ci.yml` 템플릿을 생성한다.

## Core Principles

- **프로젝트 분석**: 기존 파일 구조를 분석하여 적절한 템플릿 제안
- **모듈화**: 재사용 가능한 단계로 구성
- **환경별 설정**: dev/staging/production 분리
- **보안 고려**: 시크릿 마스킹, 최소 권한

---

## Action Detection

Parse from: $ARGUMENTS

**Supported templates**:
- `nodejs` - Node.js/npm/yarn 프로젝트
- `python` - Python/pip/poetry 프로젝트
- `go` - Go 프로젝트
- `java` - Java/Maven/Gradle 프로젝트
- `docker` - Docker 빌드 + 레지스트리 푸시
- `k8s` - Kubernetes 배포
- `custom` - 대화형으로 구성
- (empty) - 프로젝트 분석 후 자동 추천

## Template: Node.js

**Step 1: Gather Info**

```
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📦 Node.js CI/CD 설정
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

패키지 매니저:
 1. npm
 2. yarn
 3. pnpm

Node.js 버전: [예: 20, 18, 16]

테스트 명령: [기본: npm test]
빌드 명령: [기본: npm run build]
```

**Step 2: Generate Template**

```yaml
# .gitlab-ci.yml - Node.js Template
# Generated by GitLab Toolkit

stages:
  - install
  - test
  - build
  - deploy

variables:
  NODE_VERSION: "20"
  NPM_CONFIG_CACHE: "$CI_PROJECT_DIR/.npm"

cache:
  key: ${CI_COMMIT_REF_SLUG}
  paths:
    - .npm/
    - node_modules/

.node_template: &node_template
  image: node:${NODE_VERSION}
  before_script:
    - npm ci --cache .npm --prefer-offline

# Install dependencies
install:
  <<: *node_template
  stage: install
  script:
    - npm ci
  artifacts:
    paths:
      - node_modules/
    expire_in: 1 hour

# Run tests
test:
  <<: *node_template
  stage: test
  script:
    - npm test
  coverage: '/All files[^|]*\|[^|]*\s+([\d\.]+)/'
  artifacts:
    reports:
      junit: junit.xml
      coverage_report:
        coverage_format: cobertura
        path: coverage/cobertura-coverage.xml

# Lint
lint:
  <<: *node_template
  stage: test
  script:
    - npm run lint
  allow_failure: true

# Build
build:
  <<: *node_template
  stage: build
  script:
    - npm run build
  artifacts:
    paths:
      - dist/
    expire_in: 1 week
  only:
    - main
    - develop

# Deploy to staging
deploy_staging:
  stage: deploy
  script:
    - echo "Deploying to staging..."
  environment:
    name: staging
    url: https://staging.example.com
  only:
    - develop

# Deploy to production
deploy_production:
  stage: deploy
  script:
    - echo "Deploying to production..."
  environment:
    name: production
    url: https://example.com
  only:
    - main
  when: manual
```

## Template: Python

```yaml
# .gitlab-ci.yml - Python Template

stages:
  - test
  - build
  - deploy

variables:
  PIP_CACHE_DIR: "$CI_PROJECT_DIR/.pip"
  PYTHON_VERSION: "3.11"

cache:
  key: ${CI_COMMIT_REF_SLUG}
  paths:
    - .pip/
    - .venv/

.python_template: &python_template
  image: python:${PYTHON_VERSION}
  before_script:
    - python -m venv .venv
    - source .venv/bin/activate
    - pip install --upgrade pip
    - pip install -r requirements.txt

test:
  <<: *python_template
  stage: test
  script:
    - pip install pytest pytest-cov
    - pytest --cov=. --cov-report=xml
  coverage: '/TOTAL.*\s+(\d+%)/'
  artifacts:
    reports:
      coverage_report:
        coverage_format: cobertura
        path: coverage.xml

lint:
  <<: *python_template
  stage: test
  script:
    - pip install flake8 black
    - flake8 .
    - black --check .
  allow_failure: true

build:
  <<: *python_template
  stage: build
  script:
    - pip install build
    - python -m build
  artifacts:
    paths:
      - dist/
  only:
    - tags
```

## Template: Docker

```yaml
# .gitlab-ci.yml - Docker Template

stages:
  - build
  - test
  - push
  - deploy

variables:
  DOCKER_HOST: tcp://docker:2376
  DOCKER_TLS_CERTDIR: "/certs"
  IMAGE_TAG: $CI_REGISTRY_IMAGE:$CI_COMMIT_REF_SLUG
  IMAGE_TAG_LATEST: $CI_REGISTRY_IMAGE:latest

services:
  - docker:24-dind

.docker_template: &docker_template
  image: docker:24
  before_script:
    - docker login -u $CI_REGISTRY_USER -p $CI_REGISTRY_PASSWORD $CI_REGISTRY

build:
  <<: *docker_template
  stage: build
  script:
    - docker build -t $IMAGE_TAG .
    - docker push $IMAGE_TAG

test:
  <<: *docker_template
  stage: test
  script:
    - docker run --rm $IMAGE_TAG npm test
  allow_failure: true

push_latest:
  <<: *docker_template
  stage: push
  script:
    - docker pull $IMAGE_TAG
    - docker tag $IMAGE_TAG $IMAGE_TAG_LATEST
    - docker push $IMAGE_TAG_LATEST
  only:
    - main

deploy:
  stage: deploy
  image: bitnami/kubectl:latest
  script:
    - kubectl set image deployment/app app=$IMAGE_TAG
  environment:
    name: production
  only:
    - main
  when: manual
```

## Template: Kubernetes

```yaml
# .gitlab-ci.yml - Kubernetes Deployment Template

stages:
  - build
  - deploy

variables:
  KUBE_NAMESPACE: $CI_PROJECT_NAME

.kube_template: &kube_template
  image: bitnami/kubectl:latest
  before_script:
    - kubectl config set-cluster k8s --server="$KUBE_URL" --insecure-skip-tls-verify
    - kubectl config set-credentials gitlab --token="$KUBE_TOKEN"
    - kubectl config set-context default --cluster=k8s --user=gitlab --namespace=$KUBE_NAMESPACE
    - kubectl config use-context default

deploy_staging:
  <<: *kube_template
  stage: deploy
  script:
    - kubectl apply -f k8s/staging/
    - kubectl rollout status deployment/app -n $KUBE_NAMESPACE
  environment:
    name: staging
  only:
    - develop

deploy_production:
  <<: *kube_template
  stage: deploy
  script:
    - kubectl apply -f k8s/production/
    - kubectl rollout status deployment/app -n $KUBE_NAMESPACE
  environment:
    name: production
  only:
    - main
  when: manual
```
